const fs=require('node:fs'),path=require('node:path'),os=require('node:os');
const {_electron}=require('/workspace/BLEGO/node_modules/playwright');
const {imageFixture}=require('/workspace/BLEGO/tests/fixtures/images.cjs');
(async()=>{
const [root,label,out]=process.argv.slice(2);fs.mkdirSync(out,{recursive:true});
const data=fs.mkdtempSync('/tmp/blego-audit-data-');fs.writeFileSync(path.join(data,'blego-state.json'),JSON.stringify({tourSeen:true}));
const original=label==='original';
const executablePath=original?path.join(root,'node_modules/electron/dist/electron'):'/workspace/BLEGO/node_modules/electron/dist/electron';
let app;const report={label,root,at:new Date().toISOString(),platform:process.platform,arch:process.arch,window:{width:1280,height:900},externalRequests:'blocked for visual audit only',errors:[],screens:[]};
try {
app=await _electron.launch({executablePath,cwd:root,args:['--ozone-platform=headless','--disable-gpu',root,`--user-data-dir=${data}`],env:{...process.env,XDG_CACHE_HOME:'/tmp/blego-cache'},timeout:30000});
await app.context().route('**/*',r=>{const u=new URL(r.request().url()); if(u.hostname!=='127.0.0.1'||u.pathname.startsWith('/api/'))return r.abort();return r.continue();});
const page=await app.firstWindow();page.on('pageerror',e=>report.errors.push(e.message));
await app.evaluate(({BrowserWindow})=>BrowserWindow.getAllWindows()[0].setBounds({width:1280,height:900}));
await page.locator('[data-action="generate"]').first().waitFor();
await page.waitForTimeout(500);
async function capture(name){await page.waitForTimeout(200);await page.screenshot({path:path.join(out,`${label}-${name}.png`)});const info=await page.evaluate(()=>({theme:document.documentElement.dataset.theme,viewport:{width:innerWidth,height:innerHeight},controls:[...document.querySelectorAll('main button, main input, main select, main summary, nav button')].map(e=>({tag:e.tagName,label:(e.innerText||e.getAttribute('aria-label')||e.placeholder||e.name).trim().slice(0,110),action:e.dataset.action||null,id:e.id||null,type:e.type||null,disabled:e.disabled||false,visible:!!(e.offsetWidth||e.offsetHeight||e.getClientRects().length)})),animations:document.getAnimations().map(a=>({name:a.animationName||a.transitionProperty,playState:a.playState,duration:a.effect.getTiming().duration,target:a.effect.target.className})),technicalBlocks:document.querySelectorAll('pre').length})); report.screens.push({name,...info});}
await capture('generator-empty');
await page.locator('[data-action="generate"]').first().hover();await capture('generator-hover');
await page.locator('[data-action="generate"]').first().click();await capture('generator-loading');
await page.waitForTimeout(1500);await capture('generator-result');
for(const view of ['network','news','checker','photos','history','settings','import']){
const nav=page.locator(`[data-action="set-view"][data-view="${view}"]`).first();if(!await nav.count())continue;await nav.click();await capture(view);
if(view==='photos'){
const f=await imageFixture(data,'png');
if(original){const inp=page.locator('#photo-input');if(await inp.count())await inp.setInputFiles(f);}else{await app.evaluate(({dialog},f)=>{dialog.showOpenDialog=async()=>({canceled:false,filePaths:[f]});},f);await page.locator('[data-action="images-choose"]').click();}
await page.waitForTimeout(1000);await capture('photos-loaded');
}}
report.electron=await app.evaluate(()=>process.versions.electron);report.status='captured';
}catch(e){report.status='failed';report.failure=e.message;process.exitCode=1;}finally{if(app)await app.close();fs.writeFileSync(path.join(out,`${label}-audit.json`),JSON.stringify(report,null,2));fs.rmSync(data,{recursive:true,force:true});console.log(label,report.status,report.failure||'',report.errors);}
})();
